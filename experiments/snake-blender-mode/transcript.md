function updateSnakePosition(snake, velocity, eatingFood, foodValue) {
  // uses outer-scope `pendingGrowth`
  const newHead = { x: snake[0].x + velocity.x, y: snake[0].y + velocity.y };
  snake.unshift(newHead);
  if (eatingFood) pendingGrowth += (Number.isFinite(foodValue) ? foodValue : 1);
  if (pendingGrowth > 0) {
    pendingGrowth -= 1; // consume one growth unit: do NOT remove tail this tick
  } else {
    snake.pop(); // normal movement
  }
  return newHead;
}
